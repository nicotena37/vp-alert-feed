<script>
    /*
     * SOLUZIONE TELEPORT:
     * Il problema più comune nei template Joomla è overflow:hidden sull'header,
     * che taglia il dropdown-menu posizionato in absolute/fixed dentro il modulo.
     *
     * Strategia:
     * 1. Spostiamo il <ul> come figlio diretto di <body> (sfugge a qualsiasi overflow)
     * 2. Gestiamo manualmente posizionamento (fixed) e visibilità via JS
     * 3. Chiudiamo il menu al click fuori
     */
    document.addEventListener('DOMContentLoaded', function () {
        var btn = document.getElementById('btnAreaIscritti');
        var menu = document.getElementById('menuAreaIscritti');
        if (!btn || !menu) return;

        /* Teleport: sposta il menu come figlio diretto di body */
        document.body.appendChild(menu);

        var isOpen = false;

        function openMenu() {
            var rect = btn.getBoundingClientRect();
            menu.style.top = (rect.bottom + 4) + 'px';
            menu.style.left = rect.left + 'px';
            menu.style.display = 'block';
            menu.classList.add('show');
            btn.setAttribute('aria-expanded', 'true');
            isOpen = true;
        }

        function closeMenu() {
            menu.style.display = 'none';
            menu.classList.remove('show');
            btn.setAttribute('aria-expanded', 'false');
            isOpen = false;
        }

        btn.addEventListener('click', function (e) {
            e.stopPropagation();
            isOpen ? closeMenu() : openMenu();
        });

        /* Chiudi al click fuori dal menu o dal pulsante */
        document.addEventListener('click', function (e) {
            if (isOpen && !menu.contains(e.target) && e.target !== btn) {
                closeMenu();
            }
        });

        /* Ricalcola posizione al resize/scroll */
        window.addEventListener('resize', function () { if (isOpen) openMenu(); });
        window.addEventListener('scroll', function () { if (isOpen) openMenu(); }, true);
    });

		<div class="page-header">
		<h2 itemprop="headline">
															<a href="/normativa" itemprop="url"> NORMATIVA</a>
									</h2>
			</div>
			
								<p style="text-align: justify;"><strong><a href="/documenti/Normativa/Decreto%20Ministeriale%2022%20gennaio%202014%20n.%2031_Requisiti%20Organizzativi%20Mediatori.pdf">Decreto Ministero dell'Economia e delle Finanze 22 gennaio 2014, n. 31</a><br /></strong>Regolamento recante attuazione dell'articolo 29 del decreto legislativo 13 agosto 2010, n. 141, concernente il contenuto dei requisiti organizzativi per l'iscrizione nell'elenco dei mediatori creditizi.</p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/D.Lgs_169_2012.pdf" target="_blank"><strong>Decreto Legislativo 19 settembre 2012, n. 169</strong></a><br />"Ulteriori modifiche ed integrazioni al decreto legislativo 13 agosto 2010, n. 141, recante attuazione della direttiva 2008/48/CE, relativa ai contratti di credito ai consumatori, nonché modifiche del titolo V del testo unico bancario in merito alla disciplina dei soggetti operanti nel settore finanziario, degli agenti in attività finanziaria e dei mediatori creditizi."</p>
<p style="text-align: justify;"><span class="green"> </span></p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Regolamento_Agenti_IP%20(1).pdf" target="_blank"><span><strong><span>Pubblicazione del Regolamento concernente le condizioni e i requisiti per l'iscrizione nella Sezione speciale per gli agenti che prestano esclusivamente i servizi di pagamento.</span></strong></span></a></p>
<p style="text-align: justify;"><span style="line-height: 1.3em;"><span style="background-color: #ffffff; color: #000000;"><span><span>Sulla Gazzetta Ufficiale della Repubblica Italiana - Serie Generale - n. 35 dell'11 febbraio 2013, è stato pubblicato il Decreto del Ministero dell'Economia e delle Finanze del 28 dicembre 2012, n. 256, recante "Regolamento concernente le condizioni e i requisiti per l'iscrizione nella sezione speciale dell'Elenco degli Agenti in attività finanziaria da parte degli Agenti che prestano esclusivamente i servizi di pagamento". Il suddetto Regolamento entrerà in vigore a decorrere dal 26 febbraio 2013; da tale data sarà disponibile sul sito il modulo per l'istanza di iscrizione nella sezione speciale prevista dall'art. 128-quater, comma 6, del decreto legislativo 1° settembre 1993, n. 385.</span></span></span></span><span style="line-height: 1.3em;">  </span></p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Dlgs_218_10.pdf" target="_blank"><strong>Decreto Legislativo 14 dicembre 2010, n.218</strong></a><br />"Modifiche ed integrazioni al decreto legislativo 13 agosto 2010, n. 141, recante attuazione della direttiva 2008/48/CE relativa ai contratti di credito ai consumatori, nonche' modifiche del titolo IV del testo unico bancario (decreto legislativo 1° settembre 1993, n. 385) in merito alla disciplina dei soggetti operanti nel settore finanziario, degli agenti in attività finanziaria e dei mediatori creditizi”. </p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Dlgs_141_10.pdf" target="_blank"><strong>Decreto Legislativo 13 agosto 2010, n. 141</strong></a><br />Attuazione della direttiva 2008/48/CE relativa ai contratti di credito ai consumatori, nonché modifiche del titolo VI del testo unico bancario (decreto legislativo n. 385 del 1993) in merito alla disciplina dei soggetti operanti nel settore finanziario, degli agenti in attività finanziaria e dei mediatori creditizi.</p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Legge_7_luglio_2009_n_88.pdf" target="_blank"><strong>Legge 7 luglio 2009, n. 88</strong></a> (Disposizioni per l'adempimento di obblighi derivanti dall'appartenenza dell'Italia alle Comunità europee - Legge comunitaria 2008), in particolare l’art. 33“.<br />Delega al Governo per l'attuazione della direttiva 2008/48/CE del Parlamento europeo e del Consiglio, del 23 aprile 2008, relativa ai contratti di credito ai consumatori e che abroga la direttiva 87/102/CEE e previsione di modifiche ed integrazioni alla disciplina relativa ai soggetti operanti nel settore finanziario di cui al decreto legislativo 1º settembre 1993, n. 385, ai mediatori creditizi ed agli agenti in attività finanziaria”. </p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><strong><a href="/documenti/Normativa/Direttiva_2008_48_CE.pdf" target="_blank">Direttiva 2008/48/CE del Parlamento Europeo e del Consiglio del 23 aprile 2008</a></strong>, relativa ai contratti di credito ai consumatori e che abroga la direttiva 87/102/CEE </p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/L262_2005.pdf" target="_blank"><strong>Legge 28 dicembre 2005, n. 262</strong></a> - “Disposizioni per la tutela del risparmio e la disciplina dei mercati finanziari”.</p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Dlgs_485_2001.pdf" target="_blank"><strong>Decreto Ministero Economia e Finanze 13 dicembre 2001, n.485</strong></a> - “Regolamento emanato ai sensi dell'articolo 3 del decreto legislativo 25 settembre 1999, n. 374, in materia di agenzia in attività finanziaria”.</p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/DPR_28_luglio_2000_n_287.pdf" target="_blank"><strong>D.P.R. 28 luglio 2000, n. 287</strong></a>- “Regolamento di attuazione dell'articolo 6 della legge 7 marzo 1996, n. 108, recante   disciplina dell'attività di mediazione creditizia”. </p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Dlgs_374_1999.pdf" target="_blank"><strong>Decreto Legislativo 25 settembre 1999, n. 374</strong></a><br />"Estensione delle disposizioni in materia di riciclaggio dei capitali di provenienza illecita ed attività finanziarie particolarmente suscettibili di utilizzazione a fini di riciclaggio, a norma dell'articolo 15 della legge 6 febbraio 1996, n. 52”.</p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/Dlgs_153_1997.pdf" target="_blank"><strong>Decreto legislativo 26 maggio 1997, n. 153</strong></a>, convertito con modificazioni, dalla legge 5 luglio 1991, n. 197 “Integrazione dell'attuazione della Direttiva 91/308/CEE in materia di riciclaggio dei capitali di provenienza illecita”.   </p>
<p style="text-align: justify;"> </p>
<p style="text-align: justify;"><a href="/documenti/Normativa/L108_1996.pdf" target="_blank"><strong>Legge 7 marzo 1996, n. 108</strong></a> - “Disposizioni in materia di usura”.</p>
